CREATE TABLE Estoque (
	CdEstoque INT AUTO_INCREMENT,
    QtEstoque INT,
    QtMinimo INT,
    DtCompra TIMESTAMP,
    CONSTRAINT PKEstoque PRIMARY KEY (CdEstoque));
    
CREATE TABLE Produtos (
	CdProduto INT AUTO_INCREMENT,
    CdCatalago VARCHAR(50),
    NmProduto VARCHAR(100),
    VlVenda DECIMAL(10,2),
    DtVenda TIMESTAMP,
    CONSTRAINT PKProduto PRIMARY KEY (PKProduto));
    
CREATE TABLE OsProdutos (
	CdOsProduto INT AUTO_INCREMENT,
    CdAtendimento Integer,
    QtProduto Integer,
    VlProduto DECIMAL(10,2),
    DtVenda TIMESTAMP,
    CONSTRAINT PKOsProduto PRIMARY KEY (PKProduto));
    
CREATE TABLE Clientes (
	CdCliente INT AUTO_INCREMENT,
    NmCLiente VARCHAR(255),
    NrCPFCNPJ VARCHAR(16),
    NrTelefone VARCHAR(20),
    DsEndereco VARCHAR(50),
    IdAtivo VARCHAR(1),
    CONSTRAINT PKClientes PRIMARY KEY (PKClientes));
    
CREATE TABLE OS (
	CdOS INTEGER AUTO_INCREMENT,
    DTAbertura TIMESTAMP,
    DTFechamento TIMESTAMP,
    DSStatus VARCHAR(20),
    DsPagamento VARCHAR(20),
    DsObservacao VARCHAR(50),
    CdCliente INTEGER,
    NrTotal DECIMAL(10,2),
    CONSTRAINT PKOS PRIMARY KEY (CdOS));
    
CREATE TABLE Servicos (
	CdServico INT AUTO_INCREMENT,
    NmServico VARCHAR(100),
    DsServico VARCHAR(100),
    VlServico DECIMAL(10,2),
    CONSTRAINT PKServico PRIMARY KEY (CdServico));
    
CREATE TABLE OSServicoS (
	CdOSServico INT AUTO_INCREMENT,
    CdOs INTEGER,
    CdServico INTEGER,
    QTServico INTEGER,
    VlServico DECIMAL(10,2),
    DsObservacao VARCHAR(100),
    CONSTRAINT PKOSServico PRIMARY KEY (CdOSServico));
    
CREATE TABLE Caixa (
	CdCaixa INT AUTO_INCREMENT,
    DsCategoria VARCHAR(20),
    DsStatus VARCHAR(10),
    CONSTRAINT PKCaixa PRIMARY KEY (CdCaixa));
    
CREATE TABLE Categoria (
	CdCategoria INT AUTO_INCREMENT,
    NmCategoria VARCHAR(50),
    DsCategoria VARCHAR(50),
    IdAtivo VARCHAR(1),
    CONSTRAINT PKCategoria PRIMARY KEY (CdCategoria));
